    26	function findUniqueScene(project: Project, sceneId: string): Scene | undefined {
    27	  let found: Scene | undefined;
    28	  let count = 0;
    29	  for (const group of project.themeProjectGroups) {
    30	    for (const theme of group.themeProjects) {
    31	      for (const rotation of theme.rotations) {
    32	        for (const scene of rotation.scenes) {
    33	          if (scene.id === sceneId) {
    34	            found = scene;
    35	            count += 1;
    36	          }
    37	        }
    38	      }
    39	    }
    40	  }
    41	  return count === 1 ? found : undefined;
    42	}

   267	  duplicateSelection(ids: readonly string[]): MutationResult {
   268	    if (!ids.length) return { changed: false };
   269	    const selected = new Set(ids);
   270	    return this.execute("Duplicate Selection", (project) => mapProjectGroups(project, (group) => {
   271	      const themeProjects: ThemeProject[] = [];
   272	      for (const theme of group.themeProjects) {
   273	        if (selected.has(theme.id)) {
   274	          themeProjects.push(theme, duplicateThemeProject(theme));
   275	          continue;
   276	        }
   277	
   278	        const rotations: Rotation[] = [];
   279	        for (const rotation of theme.rotations) {
   280	          if (selected.has(rotation.id)) {
   281	            rotations.push(rotation, duplicateRotation(rotation));
   282	            continue;
   283	          }
   284	
   285	          const scenes: Scene[] = [];
   286	          for (const scene of rotation.scenes) {
   287	            if (selected.has(scene.id)) {
   288	              scenes.push(scene, duplicateScene(scene));
   289	              continue;
   290	            }
   291	
   292	            const widgets: Widget[] = [];
   293	            for (const widget of scene.widgets) {
   294	              widgets.push(widget);
   295	              if (selected.has(widget.id)) widgets.push(duplicateWidget(widget));
   296	            }
   297	            scenes.push({ ...scene, widgets });
   298	          }
   299	          rotations.push({ ...rotation, scenes });
   300	        }
   301	        themeProjects.push({ ...theme, rotations });
   302	      }
   303	      return { ...group, themeProjects };
   304	    }));
   305	  }

    30	export class InMemoryDocumentStore implements DocumentStore {
    31	  private currentProject: Project | undefined;
    32	  private savedProject: Project | undefined;
    33	  private snapshot: DocumentSnapshot;
    34	  private suppressSnapshotRefresh = false;
    35	  private readonly listeners = new Set<() => void>();
    36	  readonly history: CommandHistory;
    37	
    38	  constructor(history = new CommandHistory()) {
    39	    this.history = history;
    40	    this.snapshot = {
    41	      project: undefined,
    42	      isOpen: false,
    43	      isDirty: false,
    44	      history: history.snapshot,
    45	    };
    46	    this.history.subscribe(() => {
    47	      if (!this.suppressSnapshotRefresh) this.refreshSnapshot();
    48	    });
    49	  }
    50	
    51	  getCurrent(): Project | undefined { return this.currentProject; }
    52	  getSnapshot(): DocumentSnapshot { return this.snapshot; }
    53	
    54	  subscribe(listener: () => void): () => void {
    55	    this.listeners.add(listener);
    56	    return () => this.listeners.delete(listener);
    57	  }
    58	
    59	  open(project: Project): void {
    60	    this.currentProject = project;
    61	    this.savedProject = project;
    62	    this.runWithoutSnapshotRefresh(() => this.history.clear());
    63	    this.refreshSnapshot();
    64	  }
    65	
    66	  create(project: Project): void { this.open(project); }
    67	
    68	  close(): void {
    69	    this.currentProject = undefined;
    70	    this.savedProject = undefined;
    71	    this.runWithoutSnapshotRefresh(() => this.history.clear());
    72	    this.refreshSnapshot();
    73	  }
    74	
    75	  save(): void {
    76	    this.savedProject = this.currentProject;
    77	    this.refreshSnapshot();
    78	  }
    79	
    80	  replaceCurrent(project: Project): void {
    81	    if (!this.currentProject) throw new Error("No document is open");
    82	    this.currentProject = project;
    83	    if (!this.suppressSnapshotRefresh) this.refreshSnapshot();
    84	  }
    85	
    86	  execute(command: Command): void {
    87	    if (!this.currentProject) throw new Error("No document is open");
    88	    try {
    89	      this.runWithoutSnapshotRefresh(() => this.history.execute(command));
    90	    } finally {
    91	      this.refreshSnapshot();
    92	    }
    93	  }
    94	
    95	  undo(): boolean {
    96	    try {
    97	      return this.runWithoutSnapshotRefresh(() => this.history.undo());
    98	    } finally {
    99	      this.refreshSnapshot();
   100	    }
   101	  }
   102	
   103	  redo(): boolean {
   104	    try {
   105	      return this.runWithoutSnapshotRefresh(() => this.history.redo());
   106	    } finally {
   107	      this.refreshSnapshot();
   108	    }
   109	  }
   110	
   111	  private runWithoutSnapshotRefresh<T>(operation: () => T): T {
   112	    const previous = this.suppressSnapshotRefresh;
   113	    this.suppressSnapshotRefresh = true;
   114	    try {
   115	      return operation();
   116	    } finally {
   117	      this.suppressSnapshotRefresh = previous;
   118	    }
   119	  }
   120	
   121	  private refreshSnapshot(): void {
   122	    this.snapshot = {
   123	      project: this.currentProject,
   124	      isOpen: Boolean(this.currentProject),
   125	      isDirty: serialize(this.currentProject) !== serialize(this.savedProject),
   126	      history: this.history.snapshot,
   127	    };
   128	    this.listeners.forEach((listener) => listener());
   129	  }
   130	}

     7	export interface CommandHistorySnapshot {
     8	  readonly canUndo: boolean;
     9	  readonly canRedo: boolean;
    10	  readonly undoCount: number;
    11	  readonly redoCount: number;
    12	}
    13	
    14	export class CommandHistory {
    15	  private readonly undoStack: Command[] = [];
    16	  private readonly redoStack: Command[] = [];
    17	  private readonly listeners = new Set<() => void>();
    18	
    19	  subscribe(listener: () => void): () => void {
    20	    this.listeners.add(listener);
    21	    return () => this.listeners.delete(listener);
    22	  }
    23	
    24	  get snapshot(): CommandHistorySnapshot {
    25	    return {
    26	      canUndo: this.canUndo,
    27	      canRedo: this.canRedo,
    28	      undoCount: this.undoStack.length,
    29	      redoCount: this.redoStack.length,
    30	    };
    31	  }
    32	
    33	  execute(command: Command): void {
    34	    command.execute();
    35	    this.undoStack.push(command);
    36	    this.redoStack.length = 0;
    37	    this.emit();
    38	  }
    39	
    40	  undo(): boolean {
    41	    const command = this.undoStack[this.undoStack.length - 1];
    42	    if (!command) return false;
    43	
    44	    command.undo();
    45	    this.undoStack.pop();
    46	    this.redoStack.push(command);
    47	    this.emit();
    48	    return true;
    49	  }
    50	
    51	  redo(): boolean {
    52	    const command = this.redoStack[this.redoStack.length - 1];
    53	    if (!command) return false;
    54	
    55	    command.execute();
    56	    this.redoStack.pop();
    57	    this.undoStack.push(command);
    58	    this.emit();
    59	    return true;
    60	  }
    61	
    62	  clear(): void {
    63	    this.undoStack.length = 0;
    64	    this.redoStack.length = 0;
    65	    this.emit();
    66	  }
    67	
    68	  get canUndo(): boolean { return this.undoStack.length > 0; }
    69	  get canRedo(): boolean { return this.redoStack.length > 0; }
    70	
    71	  private emit(): void {
    72	    this.listeners.forEach((listener) => listener());
    73	  }
    74	}

    45	function validateCondition(
    46	  condition: Condition,
    47	  profile: DeviceProfile,
    48	  path: string,
    49	  issues: ValidationIssue[],
    50	): void {
    51	  const source = condition.source ?? "state";
    52	  const definitions = source === "setting" ? profile.runtimeSettings : profile.runtimeStates;
    53	  const definition = definitions.find((candidate) => candidate.id === condition.stateId);
    54	
    55	  if (!definition) {
    56	    issue(
    57	      issues,
    58	      "UNKNOWN_RUNTIME_REFERENCE",
    59	      `Runtime ${source} '${condition.stateId}' is not defined by the active DeviceProfile.`,
    60	      path,
    61	      "Select a DeviceProfile-defined runtime state/setting or remove the unresolved condition.",
    62	    );
    63	    return;
    64	  }
    65	
    66	  if (definition.operators && !definition.operators.includes(condition.operator)) {
    67	    issue(
    68	      issues,
    69	      "UNSUPPORTED_CONDITION_OPERATOR",
    70	      `Operator '${condition.operator}' is not supported for runtime reference '${condition.stateId}'.`,
    71	      path,
    72	      "Choose one of the operators declared by the active DeviceProfile.",
    73	    );
    74	  }
    75	
    76	  const typeValid =
    77	    definition.type === "boolean"
    78	      ? typeof condition.value === "boolean"
    79	      : definition.type === "integer"
    80	        ? typeof condition.value === "number" && Number.isInteger(condition.value)
    81	        : definition.type === "number"
    82	          ? typeof condition.value === "number" && Number.isFinite(condition.value)
    83	          : typeof condition.value === "string";
    84	
    85	  if (!typeValid) {
    86	    issue(
    87	      issues,
    88	      "INVALID_CONDITION_DATATYPE",
    89	      `Condition value for '${condition.stateId}' does not match its DeviceProfile type '${definition.type}'.`,
    90	      path,
    91	      "Use a value compatible with the runtime registry definition.",
    92	    );
    93	  }
    94	
    95	  if (definition.type === "enum" && definition.enumValues && typeof condition.value === "string" && !definition.enumValues.includes(condition.value)) {
    96	    issue(
    97	      issues,
    98	      "INVALID_CONDITION_VALUE",
    99	      `Value '${condition.value}' is not allowed for enum '${condition.stateId}'.`,
   100	      path,
   101	      "Select one of the values declared by the active DeviceProfile.",
   102	    );
   103	  }
   104	}

   282	export function validateProject(project: Project, profile?: DeviceProfile): ValidationResult {
   283	  const issues: ValidationIssue[] = [];
   284	  const assetIds = new Set(project.assets.map((asset) => asset.id));
   285	  const assets = new Map(project.assets.map((asset) => [asset.id, asset]));
   286	
   287	  if (project.name.trim().length === 0) {
   288	    issue(issues, "PROJECT_NAME_REQUIRED", "Project name is required.", "name", "Provide a non-empty project name.");
   289	  }
   290	  if (project.schemaVersion < 1) {
   291	    issue(issues, "PROJECT_SCHEMA_UNSUPPORTED", "Project schema version is not supported.", "schemaVersion", "Migrate the project to schema version 1 or newer.");
   292	  }
   293	  if (project.deviceProfileId.trim().length === 0) {
   294	    issue(issues, "DEVICE_PROFILE_REQUIRED", "A device profile must be selected.", "deviceProfileId", "Select a device profile before editing the project.");
   295	  }
   296	
   297	  const duplicateAssetIds = project.assets.filter((asset, index) => project.assets.findIndex((candidate) => candidate.id === asset.id) !== index);
   298	  uniqueIds(duplicateAssetIds).forEach((id) => issue(issues, "DUPLICATE_STABLE_ID", `Asset stable ID '${id}' is duplicated in the project scope.`, "assets", "Rename or merge the duplicated asset reference."));
   299	  project.assets.forEach((asset, index) => {
   300	    if (asset.name.trim().length === 0) issue(issues, "ASSET_NAME_REQUIRED", "Asset display name is required.", `assets[${index}].name`, "Provide a display name without changing the stable ID.");
   301	    if (asset.sourcePath.trim().length === 0) issue(issues, "ASSET_SOURCE_REQUIRED", "Asset source path is required.", `assets[${index}].sourcePath`, "Import or locate the source asset.");
   302	  });
   303	
   304	  if (!profile) {
   305	    return { valid: issues.every((current) => current.severity !== "error"), issues };
   306	  }
   307	
   308	  if (project.deviceProfileId !== profile.id) {
   309	    issue(issues, "DEVICE_PROFILE_MISMATCH", `Project profile '${project.deviceProfileId}' does not match '${profile.id}'.`, "deviceProfileId", "Resolve the active DeviceProfile before validation.");
   310	  }
   311	
   312	  project.defaultAssetIds?.forEach((assetId, index) => validateAssetReference(assetId, assets, `defaultAssetIds[${index}]`, issues));
   313	  profile.defaultAssetIds?.forEach((assetId, index) => validateAssetReference(assetId, assets, `deviceProfile.defaultAssetIds[${index}]`, issues));
   314	  project.themeProjectGroups.forEach((group, groupIndex) =>
   315	    group.themeProjects.forEach((theme, themeIndex) => validateThemeProject(theme, profile, assets, `themeProjectGroups[${groupIndex}].themeProjects[${themeIndex}]`, issues)),
   316	  );
   317	
   318	  return {
   319	    valid: issues.every((current) => current.severity !== "error"),
   320	    issues,
   321	  };
   322	}

   119	function resolveCanonicalNode(project: Project, id: string): ResolvedNode | undefined {
   120	  if (project.id === id) return { kind: "project", node: project, project };
   121	  for (const group of project.themeProjectGroups) {
   122	    if (group.id === id) return { kind: "theme-group", node: group, project, group };
   123	    for (const theme of group.themeProjects) {
   124	      if (theme.id === id) return { kind: "theme", node: theme, project, group, theme };
   125	      for (const rotation of theme.rotations) {
   126	        if (rotation.id === id) return { kind: "rotation", node: rotation, project, group, theme, rotation };
   127	        for (const scene of rotation.scenes) {
   128	          if (scene.id === id) return { kind: "scene", node: scene, project, group, theme, rotation, scene };
   129	          const widget = scene.widgets.find((candidate) => candidate.id === id);
   130	          if (widget) return { kind: "widget", node: widget, project, group, theme, rotation, scene, widget };
   131	        }
   132	      }
   133	    }
   134	  }
   135	  const asset = project.assets.find((candidate) => candidate.id === id);
   136	  return asset ? { kind: "asset", node: asset, project, asset } : undefined;

    50	export function calculateZOrderUpdates(widgets: readonly Widget[], widgetId: string, operation: ZOrderOperation): Readonly<Record<string, number>> | null {
    51	  const targetIndex = widgets.findIndex((widget) => widget.id === widgetId);
    52	  if (targetIndex < 0) return null;
    53	  const ordered = widgets.map((widget, index) => ({ widget, index })).sort((left, right) => left.widget.zIndex - right.widget.zIndex || left.index - right.index || left.widget.id.localeCompare(right.widget.id));
    54	  const orderedIndex = ordered.findIndex((entry) => entry.widget.id === widgetId);
    55	  const target = ordered[orderedIndex];
    56	  if (!target) return null;
    57	  if (operation === "bring-to-front") {
    58	    const max = Math.max(...widgets.map((widget) => widget.zIndex));
    59	    return { [widgetId]: max + 1 };
    60	  }
    61	  if (operation === "send-to-back") {
    62	    const min = Math.min(...widgets.map((widget) => widget.zIndex));
    63	    return { [widgetId]: min - 1 };
    64	  }
    65	  const neighbor = operation === "bring-forward" ? ordered[orderedIndex + 1] : ordered[orderedIndex - 1];
    66	  if (!neighbor) return null;
    67	  if (neighbor.widget.zIndex === target.widget.zIndex) return { [widgetId]: target.widget.zIndex + (operation === "bring-forward" ? 1 : -1) };
    68	  return { [widgetId]: neighbor.widget.zIndex, [neighbor.widget.id]: target.widget.zIndex };
    69	}

   255	function candidateForAxis(
   256	  axis: "x" | "y",
   257	  candidate: Geometry,
   258	  others: readonly Widget[],
   259	  configuration: SnapConfiguration,
   260	): { value: number; guide: SnapGuide; distance: number } | null {
   261	  const sourceStart = axis === "x" ? candidate.x : candidate.y;
   262	  const sourceSize = axis === "x" ? candidate.width : candidate.height;
   263	  const threshold = Math.max(0, configuration.threshold);
   264	  const gridCandidates: { value: number; guide: SnapGuide; distance: number }[] = [];
   265	  const edgeCandidates: { value: number; guide: SnapGuide; distance: number }[] = [];
   266	  const centerCandidates: { value: number; guide: SnapGuide; distance: number }[] = [];
   267	
   268	  if (configuration.grid !== false && configuration.gridSize > 0) {
   269	    const gridValue = snapValue(sourceStart, true, configuration.gridSize);
   270	    const distance = Math.abs(gridValue - sourceStart);
   271	    if (distance <= threshold) gridCandidates.push({ value: gridValue, distance, guide: { axis, position: gridValue, kind: "grid" } });
   272	  }
   273	
   274	  if (configuration.edges !== false || configuration.centers !== false) {
   275	    for (const other of others) {
   276	      const start = axis === "x" ? other.geometry.x : other.geometry.y;
   277	      const size = axis === "x" ? other.geometry.width : other.geometry.height;
   278	      const end = start + size;
   279	      if (configuration.edges !== false) {
   280	        for (const target of [start, end]) {
   281	          for (const value of [target, target - sourceSize]) {
   282	            const distance = Math.abs(value - sourceStart);
   283	            if (distance <= threshold) edgeCandidates.push({ value, distance, guide: { axis, position: target, kind: "edge", widgetId: other.id } });
   284	          }
   285	        }
   286	      }
   287	      if (configuration.centers !== false) {
   288	        const targetCenter = start + size / 2;
   289	        const value = targetCenter - sourceSize / 2;
   290	        const distance = Math.abs(value - sourceStart);
   291	        if (distance <= threshold) centerCandidates.push({ value, distance, guide: { axis, position: targetCenter, kind: "center", widgetId: other.id } });
   292	      }
   293	    }
   294	  }
   295	
   296	  for (const pass of [gridCandidates, edgeCandidates, centerCandidates]) {
   297	    if (pass.length) return pass.sort((left, right) => compareCandidates(left, right))[0];
   298	  }
   299	  return null;
   300	}
